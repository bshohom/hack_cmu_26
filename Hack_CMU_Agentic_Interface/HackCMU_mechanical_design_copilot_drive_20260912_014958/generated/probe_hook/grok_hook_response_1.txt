import numpy as np
import trimesh
from trimesh.creation import box, cylinder

PARAMS = {
    "desk_thickness_mm": 25.0,
    "clamp_gap_mm": 27.0,
    "clamp_depth_mm": 30.0,
    "clamp_width_mm": 25.0,
    "clamp_thickness_mm": 8.0,
    "vertical_drop_mm": 40.0,
    "arm_length_mm": 100.0,
    "arm_width_mm": 20.0,
    "arm_thickness_mm": 8.0,
    "tip_height_mm": 12.0,
    "wall_thickness_mm": 5.0,
}

def build(params):
    d = params
    dt = d["desk_thickness_mm"]
    cg = d["clamp_gap_mm"]
    cd = d["clamp_depth_mm"]
    cw = d["clamp_width_mm"]
    ct = d["clamp_thickness_mm"]
    vd = d["vertical_drop_mm"]
    al = d["arm_length_mm"]
    aw = d["arm_width_mm"]
    at = d["arm_thickness_mm"]
    th = d["tip_height_mm"]
    wt = d["wall_thickness_mm"]

    meshes = []

    # Upper clamp arm (x<0, z>dt)
    upper = box(extents=[cd, cw, ct], transform=trimesh.transformations.translation_matrix([-cd/2, 0, dt + ct/2]))
    meshes.append(upper)

    # Lower clamp arm (x<0, z<0)
    lower = box(extents=[cd, cw, ct], transform=trimesh.transformations.translation_matrix([-cd/2, 0, -ct/2]))
    meshes.append(lower)

    # Vertical back at edge x=0 connecting clamps
    vert = box(extents=[wt, cw, dt + cg + ct], transform=trimesh.transformations.translation_matrix([-wt/2, 0, (dt + cg + ct)/2 - ct]))
    meshes.append(vert)

    # Descent arm below desk at x~0
    descent = box(extents=[wt, aw, vd], transform=trimesh.transformations.translation_matrix([-wt/2, 0, -vd/2 - ct]))
    meshes.append(descent)

    # Horizontal arm outward +x, below desk
    horiz = box(extents=[al, aw, at], transform=trimesh.transformations.translation_matrix([al/2, 0, -vd - at/2 - ct]))
    meshes.append(horiz)

    # Raised tip at end
    tip = box(extents=[wt, aw, th], transform=trimesh.transformations.translation_matrix([al - wt/2, 0, -vd - at - th/2 - ct]))
    meshes.append(tip)

    # Union all
    part = trimesh.boolean.union(meshes, engine="manifold")
    return part

DIMENSIONS = {
    "Protrusion": 100.0,
    "Clamp gap": 27.0,
    "Hook arm length": 100.0,
    "Total height below desk": 60.0,
}

REGIONS = {

    "mounts": [
        {"name": "upper", "box": {"min": [-30.0, -12.5, 25.0], "max": [-5.0, 12.5, 33.0]}},
        {"name": "lower", "box": {"min": [-30.0, -12.5, -8.0], "max": [-5.0, 12.5, 0.0]}},
    ],
    "keep_out": [
        {"name": "throat", "box": {"min": [0.0, -10.0, -65.0], "max": [85.0, 10.0, -57.0]}},
    ],
}

NOTES = [
    "Hook for 5kg bag strap under desk edge.",
    "Clamp gap 27mm for 25mm desk.",
    "Horizontal arm 100mm out with raised tip.",
    "All walls >=5mm for FDM PLA.",
]
